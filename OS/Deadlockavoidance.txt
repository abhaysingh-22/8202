#include <iostream>
using namespace std;

int main() {
    int n, m;
    int alloc[10][10], max[10][10], need[10][10];
    int avail[10], finish[10] = {0}, safe[10];

    cout << "Enter processes and resources: ";
    cin >> n >> m;

    cout << "Enter Allocation Matrix:\n";
    for (int i = 0; i < n; i++)
        for (int j = 0; j < m; j++)
            cin >> alloc[i][j];

    cout << "Enter Max Matrix:\n";
    for (int i = 0; i < n; i++)
        for (int j = 0; j < m; j++)
            cin >> max[i][j];

    cout << "Enter Available Resources:\n";
    for (int j = 0; j < m; j++)
        cin >> avail[j];

    // Calculate Need Matrix
    for (int i = 0; i < n; i++)
        for (int j = 0; j < m; j++)
            need[i][j] = max[i][j] - alloc[i][j];

    int count = 0;
    while (count < n) {
        bool found = false;

        for (int i = 0; i < n; i++) {
            if (!finish[i]) {
                bool canRun = true;

                for (int j = 0; j < m; j++)
                    if (need[i][j] > avail[j])
                        canRun = false;

                if (canRun) {
                    for (int j = 0; j < m; j++)
                        avail[j] += alloc[i][j];

                    safe[count++] = i;
                    finish[i] = 1;
                    found = true;
                }
            }
        }

        if (!found) break;
    }

    if (count == n) {
        cout << "\nSystem is in SAFE state.\nSafe sequence: ";
        for (int i = 0; i < n; i++)
            cout << "P" << safe[i] << " ";
    } else {
        cout << "\nSystem is in UNSAFE state.";
    }

    cout << endl;
    return 0;
}
